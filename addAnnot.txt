var choice = app.popUpMenu(
    "add_Highlight_thisPage",
    "add_Cloud_thisPage",
    "rev_textSize_thisAnnot",
    "rev_textColor_thisAnnot",
    "rev_fillColor_thisAnnot",
    "saveAs_thisFile"
);

if (choice !== null) {
    switch (choice) {
        case "add_Highlight_thisPage": add_Highlight_thisPage(); break;
        case "add_Cloud_thisPage": add_Cloud_thisPage(); break;
        case "rev_textSize_thisAnnot": rev_textSize_thisAnnot(); break;
        case "rev_textColor_thisAnnot": rev_textColor_thisAnnot(); break;
        case "rev_fillColor_thisAnnot": rev_fillColor_thisAnnot(); break;
        case "saveAs_thisFile": saveAs_thisFile(); break;
    }
}

function add_Highlight_thisPage() {
    // rev 2026-10-10-h
    var vpage=this.pageNum;
    var vbox=this.getPageBox("Crop",vpage);
    this.addAnnot({
        page:vpage,
        type:"Square",
        rect:(new Matrix2D).fromRotated(this,vpage).transform([10,10,vbox[2]-10,vbox[1]-10]),
        strokeColor:["RGB",1,0,1],
        fillColor:["RGB",1,0,1],
        borderEffectStyle:"S",
        author:"U355691",
        opacity:0.30499267578125,
        subject:"highlight"
    });
}

function add_Cloud_thisPage() {
    var vpage=this.pageNum;
    vbox=this.getPageBox("Crop",vpage);
    var vbei=app.response({cQuestion:"Border Effect Intensity:",cTitle:"Curve Size",cDefault:"0.5"});
    if(vbei===null)exit;
    vbei=Number(vbei);
    var vwidth=app.response({cQuestion:"Border Width:",cTitle:"Thickness",cDefault:"0.5"});
    if(vwidth===null)exit;
    vwidth=Number(vwidth);
    this.addAnnot({
        page:vpage,
        type:"Square",
        rect:(new Matrix2D).fromRotated(this,vpage).transform([10,10,vbox[2]-10,vbox[1]-10]),
        strokeColor:color.red,
        fillColor:color.transparent,
        width:vwidth,
        borderEffectStyle:"C",
        borderEffectIntensity:vbei,
        author:"U355691",
        opacity:1,
        subject:"cloud"
    });
}

function rev_textSize_thisAnnot() {
    var s = Number(app.response("Input size:", "", String(selectedAnnots[0].getProps().richContents[0].textSize)));
    if (s > 0) {
        selectedAnnots.forEach(function (a) {
            var p = a.getProps();
            p.richContents.forEach(function (r) { r.textSize = s; });
            a.setProps(p);
        });
    }
}

function rev_textColor_thisAnnot() {
    var c = app.response("Color (name or r,g,b):", "", "red");
    if (c) {
        var col = color[c.trim().toLowerCase()] || ["RGB"].concat(c.split(",").map(Number));
        selectedAnnots.forEach(function (a) {
            var p = a.getProps();
            p.richContents.forEach(function (r) { r.textColor = col; });
            a.setProps(p);
        });
    }
}
//===============================================================================

//===============================================================================
function rev_fillColor_thisAnnot() {
    var c = app.response("Fill color (name or r,g,b):", "", "red");
    var r = c && app.response("Opacity (0-1):", "", "0.30499267578125"), o = Number(r);
    if (c && r !== null && o >= 0 && o <= 1) {
        var col = color[c.trim().toLowerCase()] || ["RGB"].concat(c.split(",").map(Number));
        selectedAnnots.forEach(function (a) {
            a.setProps({ fillColor: col, opacity: o });
        });
    }
}

function saveAs_thisFile() {
    var n = app.popUpMenu("_n1", "_o1", "_n2", "_o2", "Custom...");
    var f = n === "Custom..." ? app.response("Filename:", "Save As", "_n1") : n;

    if (f) {
        this.saveAs("/c/jo/diff/" + f + ".pdf");
        app.alert(f + ".pdf Saved!", 3);
    }
}
