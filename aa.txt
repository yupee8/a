        t.forEach(function (p) {
            var r = p.split("-");
            for (var i = +r[0]; i <= +r[r.length - 1]; i++) d[i] = 1;
        });
        for (i = 1; i <= this.numPages; i++) c.push(i);
        var o = c.filter(function (p) { return !d[p]; }),
            m = c.filter(function (p) { return d[p]; });
        j = o.indexOf(T);
        if (j < 0) return;
        o.splice.apply(o, [j + /^a/i.test(x), 0].concat(m));
        for (k = 0; k < o.length; k++) {
            j = c.indexOf(o[k]);
            j > k && this.movePage(j, k - 1);
            c.splice(k, 0, c.splice(j, 1)[0]);
        }
    }},
