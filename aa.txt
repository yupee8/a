        if(vwidth===null)exit;
        vwidth=Number(vwidth);
        this.addAnnot({
            page:vpage,
            type:"Square",
            rect:(new Matrix2D).fromRotated(this,vpage).transform([10,10,vbox[2]-10,vbox[1]-10]),

            borderEffectStyle:"C",
            borderEffectIntensity:vbei,

            subject:"cloud"

    //===========================================================================

    { name: "selAnnot_props", run: function () {
        var t = (selectedAnnots || []).map(function (a) {
            var p = a.getProps(), s = [];
            for (var k in p) s.push(k + ": " + p[k]);
            return s.sort().join("\r");
        }).join("\r\r");
        t && app.execDialog({
            initialize: function (d) { d.load({ txt1: t }); },
            description: { name: "Props", elements: [{ type: "view", elements: [
                { type: "edit_text", item_id: "txt1", multiline: true, readonly: true, font: "palette", width: 600, height: 400 },
                { type: "ok" }
            ] }] }
        });
    }},



    { name: "show_pageProps", run: function () {
        var n = this.pageNum;
        var box = this.getPageBox("Crop", n);
        var w = Math.abs(box[2] - box[0]) / 72;
        var h = Math.abs(box[3] - box[1]) / 72;

        var text = [
            "filename: " + this.documentFileName,
            "location: " + this.path,
            "filesize: " + (this.filesize / 1024).toFixed(1) + " KB",
            "page size: " + w.toFixed(2) + " x " + h.toFixed(2) + " in",
            ""
        ];

        ["Art", "Bleed", "Crop", "Media", "Trim"].forEach(function (b) {
            text.push(b + "Box: " + this.getPageBox(b, n));
        }, this);

        text.push(
            "page: " + (n + 1) + "/" + this.numPages,
            "label: " + this.getPageLabel(n),
            "rotation: " + this.getPageRotation(n),
            "words: " + this.getPageNumWords(n),
            "annots: " + (this.getAnnots({ nPage: n }) || []).length
        );

        app.execDialog({
            initialize: function (d) { d.load({ txt1: text.join("\r") }); },
            description: {
                name: "Page props",
                elements: [{ type: "view", elements: [
                    { type: "edit_text", item_id: "txt1", multiline: true, readonly: true, width: 400, height: 260 },
                    { type: "ok" }
                ] }]
            }
        });
    }},

    //===========================================================================
    { name: "rev_textSize_thisAnnot", run: function () {
        var s = Number(app.response("Input size:", "", String(selectedAnnots[0].getProps().richContents[0].textSize)));
        if (s > 0) {
            selectedAnnots.forEach(function (a) {
                var p = a.getProps();
                p.richContents.forEach(function (r) { r.textSize = s; });
                a.setProps(p);
            });
        }
    }},
    //===========================================================================


    //===========================================================================
    { name: "rev_textColor_thisAnnot", run: function () {
        var c = app.response("Color (name or r,g,b):", "", "red");
        if (c) {
            var col = color[c.trim().toLowerCase()] || ["RGB"].concat(c.split(",").map(Number));
            selectedAnnots.forEach(function (a) {
                var p = a.getProps();
                p.richContents.forEach(function (r) { r.textColor = col; });
                a.setProps(p);
            });
        }
    }},
    //===========================================================================


    //===========================================================================
    { name: "rev_fillColor_thisAnnot", run: function () {
        var c = app.response("Fill color (name or r,g,b):", "", "red");
        var r = c && app.response("Opacity (0-1):", "", "0.30499267578125"), o = Number(r);
        if (c && r !== null && o >= 0 && o <= 1) {
            var col = color[c.trim().toLowerCase()] || ["RGB"].concat(c.split(",").map(Number));
            selectedAnnots.forEach(function (a) {
                a.setProps({ fillColor: col, opacity: o });
            });
        }
    }},

    }},

    { name: "delPages", run: function () {
        var d = {}, s = app.response("Pages to delete:");
        s && s.split(",").forEach(function (p) {
            var r = p.split("-");
            for (var i = +r[0]; i <= +r[r.length - 1]; i++) d[i] = 1;
        });
        for (var i = this.numPages; i > 0; i--)
            if (d[i]) { var e = i; while (d[i - 1]) i--; this.deletePages(i - 1, e - 1); }

];
var labels = [];
for (var i = 0; i < actions.length; i++) {
    labels.push(actions[i].name);
}
var choice = app.popUpMenu.apply(app, labels);
if (choice !== null) {
    for (var j = 0; j < actions.length; j++) {
        if (actions[j].name === choice) {
            actions[j].run.call(this);
            break;
        }
    }
}
