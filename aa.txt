{ name: "setModDate", run: function () {
  var sel = this.selectedAnnots, fmt = "yyyy-mm-dd HH:MM:ss";
  if (!sel || !sel.length) return;   // nothing selected
  var s = app.response("New modDate (" + fmt + ") for " + sel.length + " selected comment(s):",
                       "Set modDate", util.printd(fmt, sel[0].modDate || new Date()));
  if (!s) return;                    // cancelled
  var d = util.scand(fmt, s);
  if (!d) { app.alert("Couldn't parse that date. Use: " + fmt); return; }
  sel.forEach(function (a) { a.modDate = d; });
}},

{ name: "movPages", run: function () {
    var s = app.response("Pages to move (5-6,B3 = before 3, 5-6,A3 = after 3):");
    if (!s) return;
    var t = s.split(","), x = t.pop(), T = +x.slice(1), d = {}, c = [], i, j, k;
    t.forEach(function (p) {
        var r = p.split("-");
        for (var i = +r[0]; i <= +r[r.length - 1]; i++) d[i] = 1;
    });
    for (i = 1; i <= this.numPages; i++) c.push(i);
    var o = c.filter(function (p) { return !d[p]; }),
        m = c.filter(function (p) { return d[p]; });
    j = o.indexOf(T);
    if (j < 0) return;
    o.splice.apply(o, [j + /^a/i.test(x), 0].concat(m));
    for (k = 0; k < o.length; k++) {
        j = c.indexOf(o[k]);
        j > k && this.movePage(j, k - 1);
        c.splice(k, 0, c.splice(j, 1)[0]);
    }
}},
